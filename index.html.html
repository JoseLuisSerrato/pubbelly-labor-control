function renderConsolBarsSVG(container,units,colors,salesArr,pptoArr){
const W=760,H=220,padL=54,padR=10,padT=10,padB=34;
const plotW=W-padL-padR,plotH=H-padT-padB;
const vals=[...salesArr,...pptoArr].filter(v=>v!=null&&v>0);
const rawMax=vals.length?Math.max(...vals):100;
const mag=Math.pow(10,Math.floor(Math.log10(rawMax||1)));
const niceMax=Math.ceil((rawMax||1)/mag)*mag||100;
const gridN=4,n=units.length,groupW=plotW/n,barW=Math.min(28,groupW*0.32),gap=6;
const y=v=>padT+plotH-(v/niceMax)*plotH;
let gridSvg='',axisLabels='';
for(let i=0;i<=gridN;i++){
const v=niceMax*i/gridN,yy=y(v);
gridSvg+=`<line x1="${padL}" y1="${yy}" x2="${W-padR}" y2="${yy}" stroke="rgba(0,0,0,.06)" stroke-width="1"/>`;
const lbl=v>=1000?`$${Math.round(v/1000)}k`:`$${Math.round(v)}`;
axisLabels+=`<text x="${padL-8}" y="${yy+4}" text-anchor="end" font-size="11" fill="#6B6B70">${lbl}</text>`;
}
let bars='',xLabels='';
units.forEach((u,i)=>{
const cx=padL+groupW*i+groupW/2;
const sv=salesArr[i],pv=pptoArr[i],color=colors[i];
if(sv!=null){
const h=Math.max(0,(sv/niceMax)*plotH),x=cx-barW-gap/2,yy=padT+plotH-h;
bars+=`<rect x="${x}" y="${yy}" width="${barW}" height="${h}" rx="4" fill="${color}CC" stroke="${color}" stroke-width="1.5"><title>${u} — Ventas reales: $${Math.round(sv).toLocaleString('en-US')}</title></rect>`;
}
if(pv!=null){
const h=Math.max(0,(pv/niceMax)*plotH),x=cx+gap/2,yy=padT+plotH-h;
bars+=`<rect x="${x}" y="${yy}" width="${barW}" height="${h}" rx="4" fill="none" stroke="#1C1C1E" stroke-width="1.5" stroke-dasharray="4,3"><title>${u} — Presupuesto: $${Math.round(pv).toLocaleString('en-US')}</title></rect>`;
}
const label=u.length>10?u.split(" ").slice(0,2).join(" "):u;
xLabels+=`<text x="${cx}" y="${H-padB+16}" text-anchor="middle" font-size="11" fill="#6B6B70">${label}</text>`;
});
container.innerHTML=`<svg viewBox="0 0 ${W} ${H}" style="width:100%;height:100%" role="img" aria-label="Barras de ventas reales vs presupuesto por unidad">${gridSvg}${axisLabels}<line x1="${padL}" y1="${padT+plotH}" x2="${W-padR}" y2="${padT+plotH}" stroke="rgba(0,0,0,.15)" stroke-width="1"/>${bars}${xLabels}</svg>`;
}
